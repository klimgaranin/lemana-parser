"""
http_utils.py — Асинхронные HTTP-запросы через curl_cffi.

curl_cffi + impersonate="chrome124" точно воспроизводит TLS-рукопожатие
Chrome 124 (JA3, ALPN, cipher suites, HTTP/2), что позволяет проходить
через Cloudflare/Qrator/DataDome даже после замены aiohttp.

После того как Playwright сделал первый запрос и получил qrator_jsid2,
curl_cffi с этим cookie и Chrome TLS-фингерпринтом работает как браузер.
"""
import asyncio
import logging
from urllib.parse import urlparse, urlunparse, quote
from typing import Optional

from curl_cffi.requests import AsyncSession

from config import CONFIG

logger = logging.getLogger("http_utils")


def _default_headers() -> dict:
    """Минимальный оверрайд поверх того что curl_cffi подставляет при chrome124."""
    h = {
        "Accept-Language": "ru-RU,ru;q=0.9,en-US;q=0.8,en;q=0.7",
        "Cache-Control":   "no-cache",
        "Pragma":          "no-cache",
    }
    if CONFIG["cookie"]:
        h["Cookie"] = CONFIG["cookie"]
    return h


def _encode_url(url: str) -> str:
    """Percent-encode Кириллицу в query-строке (libcurl требует ASCII URL)."""
    p = urlparse(url)
    return urlunparse(p._replace(query=quote(p.query, safe="=&+%")))


def create_session() -> AsyncSession:
    """
    AsyncSession с имперсонацией Chrome 124.
    Использование: async with create_session() as session: ...
    """
    return AsyncSession(
        impersonate="chrome124",
        verify=False,
        timeout=max(CONFIG["catalog_timeout"], CONFIG["product_timeout"]) + 10,
    )


async def fetch_with_retry(
    session: AsyncSession,
    url: str,
    timeout_sec: int,
    tag: str = "URL",
) -> Optional[str]:
    """GET с retry. Возвращает HTML-строку или None при исчерпании попыток."""
    safe_url = _encode_url(url)

    for attempt in range(1, CONFIG["max_retries"] + 1):
        try:
            resp = await session.get(
                safe_url,
                headers=_default_headers(),
                timeout=timeout_sec,
                allow_redirects=True,
            )
            if resp.status_code < 400:
                text = resp.text
                if text and len(text) > 200:
                    return text
                logger.warning("[%s] attempt=%d status=%d short body (%d chars)",
                               tag, attempt, resp.status_code, len(text or ""))
            else:
                logger.warning("[%s] attempt=%d HTTP %d", tag, attempt, resp.status_code)

        except asyncio.TimeoutError:
            logger.warning("[%s] attempt=%d timeout", tag, attempt)
        except Exception as exc:
            logger.warning("[%s] attempt=%d error: %s", tag, attempt, exc)

        if attempt < CONFIG["max_retries"]:
            await asyncio.sleep(CONFIG["retry_backoff"] * attempt ** 2)

    logger.error("[%s] retry exhausted: %s", tag, url)
    return None


def compute_adaptive_sleep(batch_ms: float) -> float:
    """Адаптивная пауза после батча (сек): чем быстрее сервер — тем меньше пауза."""
    lo, hi = CONFIG["min_sleep_ms"], CONFIG["max_sleep_ms"]
    if batch_ms >= 2500:
        return lo / 1000
    if batch_ms <= 800:
        return hi / 1000
    k = (2500 - batch_ms) / (2500 - 800)
    return (lo + k * (hi - lo)) / 1000

"""
catalog.py — Сбор товаров из каталожных страниц.
Параллельно тянем страницы, из каждой извлекаем items из products-list.
"""
import asyncio
import logging
import time
from typing import List, Dict

from config import CONFIG
from http_utils import fetch_with_retry, compute_adaptive_sleep
from utils import (
    strip_html, decode_html, remove_spaces, match1,
    extract_article_from_url, extract_base_url, normalize_url,
    format_fixed, _extract_price_integer_primary,
)

logger = logging.getLogger("catalog")


def _build_page_url(page: int) -> str:
    base = CONFIG["catalog_first_page_url"]
    if page <= 1:
        return base
    sep = "&" if "?" in base else "?"
    return f"{base}{sep}page={page}"


def _parse_products_count(html: str) -> int:
    import re
    m = re.search(r'data-qa-products-count=["\'](\d+)["\']', html or "", re.I)
    if m:
        return int(m.group(1))
    m = re.search(r"(\d+)\s*товар", html or "", re.I)
    return int(m.group(1)) if m else 0


def _extract_products_list_scope(html: str) -> str:
    import re
    idx = re.search(r'data-qa=["\']products-list["\']', html or "", re.I)
    if not idx:
        return ""
    return html[idx.start(): idx.start() + 250_000]


def _extract_catalog_items(html: str, base_url: str) -> List[Dict]:
    import re
    out = []
    seen_urls: set = set()

    scope = _extract_products_list_scope(html) or (html or "")

    starts = [
        (m.start(), m.group(1))
        for m in re.finditer(
            r'<div[^>]*data-qa=["\']product["\'][^>]*opp-data-product-id=["\'](\d+)["\'][^>]*>',
            scope, re.I
        )
    ]
    if not starts:
        return out

    for i, (start_idx, prod_id) in enumerate(starts):
        end_idx = starts[i + 1][0] if i + 1 < len(starts) else len(scope)
        block = scope[start_idx:end_idx]

        href = match1(block, r'href=["\'](/product/[^"\']+/)["\']')
        if not href:
            continue
        url = normalize_url(href, base_url)
        if url in seen_urls:
            continue
        seen_urls.add(url)

        name = (
            strip_html(match1(block, r'data-qa=["\']product-name["\'][^>]*>[\s\S]*?<span[^>]*>([\s\S]*?)</span>')).strip()
            or decode_html(match1(block, r'aria-label=["\']([^"\']+)["\']')).strip()
        )

        img = (
            match1(block, r'<img[^>]*itemprop=["\']image["\'][^>]*src=["\']([^"\']+)["\']')
            or match1(block, r'<img[^>]*src=["\']((https?://)[^"\']+\.(?:jpg|jpeg|png|webp)[^"\']*)["\']')
        )

        price_raw = _extract_price_integer_primary(block)
        if not price_raw:
            # fallback — последний price-integer в блоке
            for pm in re.finditer(
                r'data-testid=["\']price-integer["\'][^>]*>([\s\S]*?)</span>', block, re.I
            ):
                price_raw = remove_spaces(decode_html(strip_html(pm.group(1))))
        price = format_fixed(price_raw, 2)

        out.append({
            "article": prod_id or extract_article_from_url(url),
            "url":     url,
            "name":    name,
            "price":   price,
            "image":   img or "",
        })

    return out


async def _fetch_page(session, page: int, base_url: str, sem: asyncio.Semaphore) -> List[Dict]:
    async with sem:
        url = _build_page_url(page)
        html = await fetch_with_retry(session, url, CONFIG["catalog_timeout"], f"CATALOG_{page}")
        if not html:
            return []
        return _extract_catalog_items(html, base_url)


async def collect_catalog_items(session) -> List[Dict]:
    base_url = extract_base_url(CONFIG["catalog_first_page_url"])
    items_map: Dict[str, Dict] = {}

    # ── Page 1 ───────────────────────────────────────────────────────────────
    html1 = await fetch_with_retry(session, _build_page_url(1), CONFIG["catalog_timeout"], "CATALOG_1")
    if not html1:
        raise RuntimeError("Каталог page=1 недоступен. Проверь URL в config.py")

    total_count = _parse_products_count(html1)
    items1 = _extract_catalog_items(html1, base_url)

    if not items1:
        raise RuntimeError(
            "На page=1 товаров не найдено. "
            "Playwright мог не дождаться загрузки — увеличь CHALLENGE_WAIT_SEC."
        )

    per_page = len(items1)
    for it in items1:
        items_map[it["url"]] = it

    total_pages = min(
        (((total_count - 1) // per_page) + 1) if total_count else CONFIG["max_pages_safety"],
        CONFIG["max_pages_safety"],
    )
    logger.info("ℹ️  totalCount=%d, perPage=%d, totalPages=%d", total_count, per_page, total_pages)

    if total_pages <= 1:
        return list(items_map.values())[: CONFIG["max_products"]]

    # ── Pages 2..N параллельно ───────────────────────────────────────────────
    sem = asyncio.Semaphore(CONFIG["catalog_concurrency"])
    pages = list(range(2, total_pages + 1))

    from tqdm import tqdm
    with tqdm(total=len(pages), desc="📄 Каталог (страниц)", unit="стр") as pbar:
        batch_size = CONFIG["catalog_concurrency"] * 2

        for start in range(0, len(pages), batch_size):
            batch = pages[start: start + batch_size]
            t0 = time.monotonic()

            results = await asyncio.gather(*[
                _fetch_page(session, p, base_url, sem) for p in batch
            ])

            for page_items in results:
                for it in page_items:
                    if it["url"] not in items_map:
                        items_map[it["url"]] = it

            pbar.update(len(batch))
            pbar.set_postfix({"товаров": len(items_map)})

            if len(items_map) >= CONFIG["max_products"]:
                break

            elapsed_ms = (time.monotonic() - t0) * 1000 / max(len(batch), 1)
            sleep_sec = compute_adaptive_sleep(elapsed_ms)
            if sleep_sec > 0:
                await asyncio.sleep(sleep_sec)

    return list(items_map.values())[: CONFIG["max_products"]]

"""
products.py — Параллельная загрузка и парсинг карточек товаров.
"""
import asyncio
import logging
import time
from typing import List, Dict, Tuple

from config import CONFIG
from http_utils import fetch_with_retry, compute_adaptive_sleep
from utils import (
    strip_html, match1, extract_article_from_url,
    parse_price_from_html, extract_main_image,
    extract_all_characteristics,
)

logger = logging.getLogger("products")


def _parse_product(html: str, base: Dict) -> Dict:
    result = {
        "article": base.get("article") or extract_article_from_url(base["url"]),
        "url":     base["url"],
        "name":    base.get("name", ""),
        "price":   base.get("price", ""),
        "image":   base.get("image", ""),
        "characteristics": {},
    }

    if not result["name"]:
        h1 = match1(html, r"<h1[^>]*>([\s\S]*?)</h1>")
        result["name"] = strip_html(h1).strip()

    if not result["price"]:
        result["price"] = parse_price_from_html(html)

    if not result["image"]:
        result["image"] = extract_main_image(html)

    result["characteristics"] = extract_all_characteristics(html)
    return result


async def _fetch_product(session, item: Dict, sem: asyncio.Semaphore) -> Dict:
    async with sem:
        tag = f"PROD_{item.get('article', item['url'][-20:])}"
        html = await fetch_with_retry(session, item["url"], CONFIG["product_timeout"], tag)
        if not html:
            return {
                "article": item.get("article", ""),
                "url":     item["url"],
                "name":    item.get("name", ""),
                "price":   item.get("price", ""),
                "image":   item.get("image", ""),
                "characteristics": {},
            }
        return _parse_product(html, item)


async def fetch_and_parse_products(
    session,
    catalog_items: List[Dict],
) -> Tuple[List[Dict], List[str]]:
    sem = asyncio.Semaphore(CONFIG["product_concurrency"])
    products: List[Dict] = []
    all_char_keys: Dict[str, bool] = {}

    from tqdm import tqdm
    with tqdm(total=len(catalog_items), desc="🔎 Карточки товаров", unit="шт") as pbar:
        batch_size = CONFIG["product_concurrency"] * 2

        for start in range(0, len(catalog_items), batch_size):
            batch = catalog_items[start: start + batch_size]
            t0 = time.monotonic()

            results = await asyncio.gather(*[
                _fetch_product(session, item, sem) for item in batch
            ])

            for product in results:
                for key in product["characteristics"]:
                    all_char_keys[key] = True
                products.append(product)

            pbar.update(len(batch))
            elapsed_ms = (time.monotonic() - t0) * 1000 / max(len(batch), 1)
            sleep_sec = compute_adaptive_sleep(elapsed_ms)
            if sleep_sec > 0:
                await asyncio.sleep(sleep_sec)

    char_keys_sorted = sorted(all_char_keys.keys(), key=lambda x: x.lower())
    return products, char_keys_sorted
